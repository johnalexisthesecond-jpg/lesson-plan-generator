function getLines(value){
    return value
        .split("\n")
        .map(v => v.trim())
        .filter(v => v !== "");
}

function validate(data){

    for(let item of data){

        if(item.trim() === ""){
            return false;
        }

    }

    return true;
}

function generateObjectives(subject){

    let objectives = "";

    const items = [
        `Define major concepts related to ${subject}.`,
        `Explain important principles in ${subject}.`,
        `Analyze situations connected to ${subject}.`,
        `Apply concepts in real-world contexts.`,
        `Develop evidence-based conclusions.`,
        `Collaborate during learning activities.`,
        `Use critical thinking in solving problems.`,
        `Communicate ideas effectively.`,
        `Reflect on learning experiences.`,
        `Demonstrate mastery through performance tasks.`
    ];

    items.forEach(i => {
        objectives += `<li>${i}</li>`;
    });

    return `<ol>${objectives}</ol>`;
}

function createChecklist(count, text){

    let html = "<ul>";

    for(let i=1; i<=count; i++){

        html += `<li>${text} ${i}</li>`;

    }

    html += "</ul>";

    return html;
}
